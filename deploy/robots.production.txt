# solsticellp.com - the live site. Crawlable, deliberately.
#
# The staging host serves deploy/robots.staging.txt instead, which is
# Disallow: /. Which file a build gets is the ROBOTS_SRC build arg in
# deploy/web.Dockerfile - production must never inherit the staging one, and
# that mistake is exactly what this split exists to prevent.

User-agent: *
Allow: /

# The admin has been a real path since the hash router went. The control is
# the X-Robots-Tag nginx sends on /admin (deploy/nginx.conf); this line only
# saves crawlers the fetch. It used to read /#admin, which matched nothing -
# a fragment is never part of the path robots.txt rules are compared against.
Disallow: /admin

# Written by the prerender job from the exact set of pages it rendered, so it
# lists only published pages and real products. Absent until the job has run
# once, in which case this line points at a 404 - harmless, and fixed by the
# first run.
Sitemap: https://solsticellp.com/sitemap.xml
